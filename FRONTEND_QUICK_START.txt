  REACT:
  ─────
  import { useEffect, useState } from 'react';

  function RouteMap({ routeId }) {
    const [route, setRoute] = useState(null);

    useEffect(() => {
      fetch(`http://localhost:3000/routes/${routeId}/optimized`, {
        headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
      })
        .then(r => r.json())
        .then(setRoute);
    }, [routeId]);

    if (!route) return <div>Cargando...</div>;

    return (
      <div>
        <h2>🗺️ {route.totalDistanceKm} km</h2>
        <ol>
          {route.optimizedRoute.map((p, i) => (
            <li key={i}>{i+1}. {p.name}</li>
          ))}
        </ol>
      </div>
    );
  }


  VUE:
  ────
  <template>
    <div>
      <h2>🗺️ {{ route?.totalDistanceKm }} km</h2>
      <ol>
        <li v-for="(p, i) in route?.optimizedRoute" :key="i">
          {{ i+1 }}. {{ p.name }}
        </li>
      </ol>
    </div>
  </template>

  <script setup>
  import { ref, onMounted } from 'vue';
  const route = ref(null);
  onMounted(async () => {
    const r = await fetch(
      `http://localhost:3000/routes/{routeId}/optimized`,
      { headers: { Authorization: `Bearer ${localStorage.getItem('token')}` } }
    );
    route.value = await r.json();
  });
  </script>

┌─────────────────────────────────────────────────────────────────────────────┐
│ 8. TROUBLESHOOTING                                                           │
└─────────────────────────────────────────────────────────────────────────────┘

  ❌ "Error: Unauthorized" en consola
     ✅ Verificar que token está en localStorage
     ✅ Verificar que no está expirado

  ❌ Mapa no aparece
     ✅ Verificar API Key en .env
     ✅ Verificar que Google Maps esté habilitado en Google Cloud

  ❌ Marcadores no se ven
     ✅ Verificar que lat/lng sean números válidos
     ✅ Verificar que el zoom sea correcto (12-15)

  ❌ Datos tardan mucho
     ✅ Verificar conexión a internet
     ✅ Verificar que backend esté corriendo
     ✅ Revisar console.log() para ver respuesta

  ❌ Ruta no se ve en mapa
     ✅ Verificar que optimizedRoute tiene datos
     ✅ Verificar console.log(route)

│ 9. PREGUNTAS RÁPIDAS                                                        │
└─────────────────────────────────────────────────────────────────────────────┘

  P: ¿Debo refrescar cada X segundos?
  R: No. Los datos son estáticos. Solo actualiza si hay cambios.

  P: ¿Puedo mostrar rastreo en vivo?
  R: Sí, pero necesitas WebSocket (futuro, más adelante).

  P: ¿Los datos se guardan en mi app?
  R: Sí, guarda en Redux/Context para que persista entre páginas.

  P: ¿Debo pedir confirmación antes de mostrar mapa?
  R: No. Solo verifica que la ruta esté iniciada.

  P: ¿Qué pasa si el usuario no tiene Google Maps API Key?
  R: El mapa no aparecerá, pero la lista de paradas sí.
